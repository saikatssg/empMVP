For production, the frontend is built natively by Vite (npm run build) for static hosting (Vercel/Netlify), while the backend module is isolated for deploying to cloud environments like AWS ECS, EC2, or Heroku.

1. Auth Controller (backend/src/modules/auth/auth.controller.js)

This controller manages secure access. Instead of sending the JWT in the JSON body (which is vulnerable to Cross-Site Scripting), it attaches the token to an httpOnly cookie.

import jwt from 'jsonwebtoken';
import bcrypt from 'bcryptjs';
import User from '../users/user.model.js';

/**
 * @desc    Authenticate user & get token
 * @route   POST /api/auth/login
 * @access  Public
 */
export const login = async (req, res) => {
  try {
    const { email, password } = req.body;

    // 1. Check if user exists (explicitly select password field since it's hidden by default)
    const user = await User.findOne({ email }).select('+passwordHash');
    if (!user || !user.isActive) {
      return res.status(401).json({ message: 'Invalid credentials or inactive account' });
    }

    // 2. Verify password
    const isMatch = await bcrypt.compare(password, user.passwordHash);
    if (!isMatch) {
      return res.status(401).json({ message: 'Invalid credentials' });
    }

    // 3. Generate JWT
    const token = jwt.sign(
      { userId: user._id, role: user.role },
      process.env.JWT_SECRET,
      { expiresIn: '8h' }
    );

    // 4. Set HTTP-Only Cookie
    res.cookie('token', token, {
      httpOnly: true, // Prevents client-side JS from reading the cookie (XSS protection)
      secure: process.env.NODE_ENV === 'production', // Requires HTTPS in production
      sameSite: 'strict', // CSRF protection
      maxAge: 8 * 60 * 60 * 1000, // 8 hours
    });

    // Remove password from response
    user.passwordHash = undefined;

    res.status(200).json({
      message: 'Login successful',
      user: {
        id: user._id,
        email: user.email,
        role: user.role,
      }
    });
  } catch (error) {
    res.status(500).json({ message: 'Server error during login', error: error.message });
  }
};

/**
 * @desc    Logout user & clear cookie
 * @route   POST /api/auth/logout
 * @access  Private
 */
export const logout = async (req, res) => {
  res.cookie('token', '', {
    httpOnly: true,
    expires: new Date(0), // Instantly expires the cookie
  });
  res.status(200).json({ message: 'Logged out successfully' });
};


2. Employee Controller (backend/src/modules/employees/employee.controller.js)

This controller handles the core business logic. The createEmployee function uses MongoDB Transactions to guarantee that if the Employee profile fails to save, the User authentication account is rolled back, preventing orphaned data.

import mongoose from 'mongoose';
import bcrypt from 'bcryptjs';
import Employee from './employee.model.js';
import User from '../users/user.model.js';

/**
 * @desc    Create a new employee (Admin only)
 * @route   POST /api/employees
 * @access  Private/Admin
 */
export const createEmployee = async (req, res) => {
  // Start a transaction session
  const session = await mongoose.startSession();
  session.startTransaction();

  try {
    const { email, password, role, firstName, lastName, phone, departmentId, managerId } = req.body;

    // 1. Check if user email already exists
    const existingUser = await User.findOne({ email }).session(session);
    if (existingUser) {
      throw new Error('Email is already registered');
    }

    // 2. Hash the initial password
    const salt = await bcrypt.genSalt(12);
    const passwordHash = await bcrypt.hash(password, salt);

    // 3. Create the User record (Auth entity)
    const [newUser] = await User.create([{ email, passwordHash, role }], { session });

    // 4. Create the Employee record (Profile entity) linked to the User
    const [newEmployee] = await Employee.create([{
      userId: newUser._id,
      firstName,
      lastName,
      phone,
      departmentId,
      managerId: managerId || null
    }], { session });

    // 5. Commit the transaction (Saves both to DB simultaneously)
    await session.commitTransaction();
    session.endSession();

    res.status(201).json({
      message: 'Employee provisioned successfully',
      employee: newEmployee
    });

  } catch (error) {
    // If anything fails, abort the transaction so no partial data is saved
    await session.abortTransaction();
    session.endSession();
    res.status(400).json({ message: 'Failed to create employee', error: error.message });
  }
};

/**
 * @desc    Get all employees with filtering, searching, and populating
 * @route   GET /api/employees
 * @access  Private
 */
export const getEmployees = async (req, res) => {
  try {
    const { search, department, managerId } = req.query;
    
    // Build dynamic query object based on frontend filters
    const query = {};

    if (department) {
      query.departmentId = department;
    }
    
    if (managerId) {
      query.managerId = managerId;
    }

    if (search) {
      // Allow searching by first or last name using regex
      query.$or = [
        { firstName: { $regex: search, $options: 'i' } }, // 'i' makes it case-insensitive
        { lastName: { $regex: search, $options: 'i' } }
      ];
    }

    // Execute query and populate related department and manager names
    const employees = await Employee.find(query)
      .populate('departmentId', 'name budgetCode') // Only fetch department name and budget code
      .populate('managerId', 'firstName lastName') // Only fetch manager's name
      .populate('userId', 'email role isActive') // Fetch auth details
      .sort({ lastName: 1 }); // Alphabetical sort by last name

    res.status(200).json({
      count: employees.length,
      data: employees
    });
  } catch (error) {
    res.status(500).json({ message: 'Server Error fetching employees', error: error.message });
  }
};

1. Authentication Middleware (backend/src/middleware/auth.middleware.js)

This middleware verifies the JWT and attaches the full user object to the req object. It also acts as a security check to ensure the user hasn't been deactivated or deleted since the token was issued.

import jwt from 'jsonwebtoken';
import User from '../modules/users/user.model.js';

export const protectRoute = async (req, res, next) => {
  try {
    // Extract token from the httpOnly cookie
    const token = req.cookies.token;

    if (!token) {
      return res.status(401).json({ message: 'Access denied. No token provided.' });
    }

    // Verify token validity and expiration
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    // Fetch the user from the database to ensure they still exist and are active
    const currentUser = await User.findById(decoded.userId);
    
    if (!currentUser) {
      return res.status(401).json({ message: 'The user belonging to this token no longer exists.' });
    }

    if (!currentUser.isActive) {
      return res.status(403).json({ message: 'This account has been deactivated.' });
    }

    // Attach user to the request object for use in downstream controllers
    req.user = currentUser;
    next();
    
  } catch (error) {
    if (error.name === 'TokenExpiredError') {
      return res.status(401).json({ message: 'Session expired. Please log in again.' });
    }
    return res.status(401).json({ message: 'Invalid token.', error: error.message });
  }
};


2. Role-Based Access Control (RBAC) Middleware (backend/src/middleware/role.middleware.js)

This middleware is executed after the authentication middleware. It checks if the authenticated user's role matches the roles allowed to access the specific route.

export const restrictTo = (...allowedRoles) => {
  return (req, res, next) => {
    // req.user is populated by the protectRoute middleware
    if (!req.user || !allowedRoles.includes(req.user.role)) {
      return res.status(403).json({ 
        message: `Forbidden: Your role (${req.user.role}) lacks permissions for this action.` 
      });
    }
    
    next();
  };
};


How to Apply Them in Your Routes

Here is how you chain these middlewares inside your route definitions (backend/src/modules/employees/employee.routes.js) to secure the endpoints:

import express from 'express';
import { createEmployee, getEmployees } from './employee.controller.js';
import { protectRoute } from '../../middleware/auth.middleware.js';
import { restrictTo } from '../../middleware/role.middleware.js';

const router = express.Router();

// Apply protectRoute to all employee endpoints so only logged-in users can access them
router.use(protectRoute);

// Standard users can view the directory, but only Admins can create new employees
router.route('/')
  .get(getEmployees) // Any authenticated user can access
  .post(restrictTo('Admin'), createEmployee); // Blocked for standard users

export default router;

User prompt: How do I configure my React Vite frontend and Axios instance to ensure these HTTP-only cookies are sent back to the API on every request?

Response: To successfully pass HTTP-only cookies back and forth between a decoupled React frontend and a Node API, you must explicitly tell Axios to attach credentials to every request. By default, browsers block cookies on cross-origin requests for security.

Here is the complete configuration required across your Axios instance, Vite dev server, and Express backend.

1. Configure the Axios Instance (Frontend)

Instead of importing axios directly in your components, create a centralized instance. The critical setting here is withCredentials: true.

Create frontend/src/config/axiosInstance.js:

import axios from 'axios';

// Create a customized Axios instance
const apiClient = axios.create({
  // If using the Vite proxy (step 2), this can just be '/api'
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  
  // CRITICAL: This tells the browser to include the HTTP-Only cookie in the request
  withCredentials: true,
  
  headers: {
    'Content-Type': 'application/json',
  },
});

// Response Interceptor: Globally handle expired sessions
apiClient.interceptors.response.use(
  (response) => {
    // Return successful responses unmodified
    return response;
  },
  (error) => {
    // If the backend returns a 401 Unauthorized (e.g., token expired or invalid)
    if (error.response && error.response.status === 401) {
      // Clear local user state and redirect to login
      // Note: You might want to integrate this with your AuthContext
      console.warn('Session expired. Redirecting to login.');
      window.location.href = '/login'; 
    }
    
    return Promise.reject(error);
  }
);

export default apiClient;


How to use it in your service files:

export const getEmployees = async () => {
  // No need to manually attach a token; the browser sends the cookie automatically
  const response = await apiClient.get('/employees');

};

During local development, your frontend runs on a different port (e.g., 5173) than your backend (e.g., 5000). This triggers strict CORS (Cross-Origin Resource Sharing) browser policies.

In your backend/src/app.js:

2. The AuthContext Implementation

This file creates the Context, the Provider component, and a custom useAuth hook for easy consumption across your application.

Create frontend/src/context/AuthContext.jsx:

import { createContext, useState, useEffect, useContext } from 'react';
import apiClient from '../config/axiosInstance';

// 1. Create the context
const AuthContext = createContext();

// 2. Create the Provider component
export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true); // Prevents rendering protected routes before checking session

  // Check if user is logged in when the app first loads or refreshes
  useEffect(() => {
    const checkAuthStatus = async () => {
      try {
        const response = await apiClient.get('/auth/me');
        setUser(response.data);
      } catch (error) {
        // If 401 Unauthorized, the user is not logged in or cookie expired
        setUser(null);
      } finally {
        setLoading(false); // App is ready to render
      }
    };

    checkAuthStatus();
  }, []);

  // Login action
  const login = async (email, password) => {
    const response = await apiClient.post('/auth/login', { email, password });
    setUser(response.data.user);
    return response.data; // Return data so the UI can show success messages
  };

  // Logout action
  const logout = async () => {
    await apiClient.post('/auth/logout');
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, isAuthenticated: !!user, loading, login, logout }}>
      {/* Do not render children (the app) until we know the auth status */}
      {!loading && children}
    </AuthContext.Provider>
  );
};

// 3. Custom Hook for easy importing
export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
};

User prompt: How do I build the PrivateRoute and AdminRoute components to protect my React pages using this AuthContext?

Response: Using React Router v6, route protection is handled cleanly using wrapper components. These components consume your AuthContext and decide whether to render the requested page (via the <Outlet/> component) or redirect the user elsewhere.

Here is how to build and implement both the PrivateRoute and AdminRoute.

1. Build the PrivateRoute Component

This component ensures that only logged-in users can access standard application routes like their profile or the employee directory.

Create frontend/src/router/PrivateRoute.jsx:

import { Navigate, Outlet, useLocation } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';

const PrivateRoute = () => {
  const { isAuthenticated } = useAuth();
  const location = useLocation();

  // If the user is logged in, render the child routes (Outlet)
  // If not, redirect to login and save the location they were trying to access
  return isAuthenticated ? (
    <Outlet />
  ) : (
    <Navigate to="/login" state={{ from: location }} replace />
  );
};

2. Build the AdminRoute Component

This component adds a second layer of checks. It ensures the user is both logged in and holds the 'Admin' role before granting access to sensitive routes like employee creation or system analytics.

Create frontend/src/router/AdminRoute.jsx:

import { Navigate, Outlet } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';

const AdminRoute = () => {
  const { isAuthenticated, user } = useAuth();

  // If not logged in, boot them to the login page
  if (!isAuthenticated) {
    return <Navigate to="/login" replace />;
  }

  // If logged in but NOT an Admin, boot them to the standard dashboard
  if (user?.role !== 'Admin') {
    return <Navigate to="/dashboard" replace />;
  }

  // If both checks pass, render the protected admin routes
  return <Outlet />;
};

export default AdminRoute;

Employee Schema (With Auto-ID Generation Hook)

This uses a Mongoose pre('save') hook to automatically generate the ID ([DeptName]-[MMYYYY]-[001]) right before the document is saved to the database.

// backend/src/modules/employees/employee.model.js
import mongoose from 'mongoose';

const employeeSchema = new mongoose.Schema({
  userId: { type: mongoose.Schema.Types.ObjectId, ref: 'User', required: true },
  employeeId: { type: String, unique: true }, // e.g., IT-102026-001
  firstName: { type: String, required: true },
  lastName: { type: String, required: true },
  departmentId: { type: mongoose.Schema.Types.ObjectId, ref: 'Department', required: true },
}, { timestamps: true });

// Pre-save hook to generate the custom Employee ID
employeeSchema.pre('save', async function (next) {
  // Only generate ID if this is a newly created document
  if (!this.isNew) return next();

  try {
    // 1. Get the Department Name
    const dept = await mongoose.model('Department').findById(this.departmentId);
    if (!dept) throw new Error('Invalid Department ID');
    
    // Format Dept Name (Remove spaces, uppercase)
    const formattedDept = dept.name.toUpperCase().replace(/\s+/g, '');

    // 2. Get Current Month & Year (e.g., 102026)
    const now = new Date();
    const monthYear = `{now.getFullYear()}`;

    // 3. Create the prefix (e.g., IT-102026-)
    const prefix = `{monthYear}-`;

    // 4. Find the last employee created in this department this month
    const lastEmployee = await this.constructor
      .findOne({ employeeId: new RegExp(`^${prefix}`) })
      .sort({ employeeId: -1 });

    // 5. Calculate Sequence (001, 002, etc.)
    let sequence = 1;
    if (lastEmployee && lastEmployee.employeeId) {
      const lastSeq = parseInt(lastEmployee.employeeId.split('-')[2], 10);
      sequence = lastSeq + 1;
    }

    // 6. Assign the final ID
    this.employeeId = `${prefix}${String(sequence).padStart(3, '0')}`;
    next();
  } catch (error) {
    next(error);
  }
});

export default mongoose.model('Employee', employeeSchema);

    // SuperAdmins bypass all checks
    if (req.user.role === 'SuperAdmin') {

3. Two-Step Verification (2FA) Auth Controller

The login flow is now broken into two steps:

Validate credentials -> Generate & Send OTP.

Validate OTP -> Issue HTTP-Only JWT Cookie.

// backend/src/modules/auth/auth.controller.js
import crypto from 'crypto';
import User from '../users/user.model.js';
import { sendEmail } from '../../utils/mailer.js'; // Assumes a Nodemailer/SendGrid wrapper

/**
 * Step 1: Login & Trigger OTP
 */
export const loginInitiate = async (req, res) => {
  const { email, password } = req.body;
  const user = await User.findOne({ email }).select('+passwordHash');
  
  if (!user || !(await bcrypt.compare(password, user.passwordHash))) {
    return res.status(401).json({ message: 'Invalid credentials' });
  }

  // Generate 6-digit OTP
  const otp = Math.floor(100000 + Math.random() * 900000).toString();
  
  // Hash OTP before saving to DB for security
  const hashedOtp = crypto.createHash('sha256').update(otp).digest('hex');
  
  user.authOtp = hashedOtp;
  user.authOtpExpires = Date.now() + 10 * 60 * 1000; // 10 minutes expiry
  await user.save();

  // Send OTP via Email (or Twilio for SMS)
  await sendEmail({
    to: user.email,
    subject: 'Your 2FA Login Code',
    text: `Your verification code is: ${otp}. It expires in 10 minutes.`
  });

  res.status(200).json({ 
    message: 'Step 1 complete. 2FA code sent to your registered device/email.',
    requires2FA: true,
    email: user.email 
  });
};

/**
 * Step 2: Verify OTP & Issue Session
 */
export const loginVerify = async (req, res) => {
  const { email, otp } = req.body;
  
  const user = await User.findOne({ email }).select('+authOtp +authOtpExpires');
  if (!user) return res.status(401).json({ message: 'Invalid request' });

  // Verify OTP validity and expiry
  const hashedInputOtp = crypto.createHash('sha256').update(otp).digest('hex');
  
  if (user.authOtp !== hashedInputOtp || user.authOtpExpires < Date.now()) {
    return res.status(400).json({ message: 'Invalid or expired 2FA code' });
  }

  // Clear OTP fields
  user.authOtp = undefined;
  user.authOtpExpires = undefined;
  await user.save();

  // Issue HTTP-Only Cookie (as defined in previous steps)
  const token = jwt.sign({ userId: user._id, role: user.role }, process.env.JWT_SECRET, { expiresIn: '8h' });
  
  res.cookie('token', token, { httpOnly: true, secure: true, sameSite: 'strict', maxAge: 8 * 60 * 60 * 1000 });
  res.status(200).json({ message: 'Login successful' });
};

// PUT /api/auth/update-credentials
export const updateCredentials = async (req, res) => {
  // req.user is provided by the protectRoute middleware
  const { newPassword, otp } = req.body;
  
  // 1. Verify the OTP provided matches the one sent to their active session
  const hashedInputOtp = crypto.createHash('sha256').update(otp).digest('hex');
  
  const user = await User.findById(req.user._id).select('+authOtp +authOtpExpires');
  if (user.authOtp !== hashedInputOtp || user.authOtpExpires < Date.now()) {
    return res.status(403).json({ message: 'Update failed: Invalid verification code.' });
  }

  // 2. Hash and save the new password
  const salt = await bcrypt.genSalt(12);
  user.passwordHash = await bcrypt.hash(newPassword, salt);
  
  // Clear the OTP
  user.authOtp = undefined;
  user.authOtpExpires = undefined;
  await user.save();

  res.status(200).json({ message: 'Credentials updated successfully. Next login will require the new password.' });
};

import { createContext, useState, useEffect, useContext } from 'react';
import apiClient from '../config/axiosInstance';

const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  // Check auth status on load
  useEffect(() => {
    const checkAuthStatus = async () => {
      try {
        const response = await apiClient.get('/auth/me');
        setUser(response.data);
      } catch (error) {
        setUser(null);
      } finally {
        setLoading(false);
      }
    };
    checkAuthStatus();
  }, []);

  // Step 1: Validate credentials and trigger OTP email/SMS
  const loginInitiate = async (email, password) => {
    const response = await apiClient.post('/auth/login', { email, password });
    return response.data; // Returns { requires2FA: true, message: '...' }
  };

  // Step 2: Verify OTP and fetch user profile
  const loginVerify = async (email, otp) => {
    // 1. Send OTP to backend to set the HTTP-Only cookie
    await apiClient.post('/auth/verify-otp', { email, otp });
    
    // 2. Fetch the user profile now that the cookie is set
    const userResponse = await apiClient.get('/auth/me');
    setUser(userResponse.data);
  };

  const logout = async () => {
    await apiClient.post('/auth/logout');
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ 
      user, 
      isAuthenticated: !!user, 
      loading, 
      loginInitiate, 
      loginVerify, 
      logout 
    }}>
      {!loading && children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth must be used within an AuthProvider');
  return context;
};
